import pandas as pd
import polars as pl
import yaml
import itertools
import os

with open("calkit.yaml") as f:
    calkit_config = yaml.safe_load(f)

# Load the BOOM and Kowalski configurations
boom_params = {}
for i in calkit_config["pipeline"]["stages"]["benchmark-boom"]["iterate_over"]:
    boom_params[i["arg_name"]] = i["values"]

kowalski_params = {}
for i in calkit_config["pipeline"]["stages"]["benchmark-kowalski"][
    "iterate_over"
]:
    kowalski_params[i["arg_name"]] = i["values"]

# Form the matrix of parameters for each
param_names = list(boom_params.keys())
param_vals = list(boom_params.values())
vals_product = list(itertools.product(*param_vals))
for i in calkit_config["pipeline"]["stages"]["benchmark-boom-more"][
    "iterate_over"
][0]["values"]:
    vals_product.append(tuple(i))
results_boom = []

for vals in vals_product:
    current_params = dict(zip(param_names, vals))
    boom_config = (
        f"na={current_params['n_alert_workers']}-"
        f"ne={current_params['n_enrichment_workers']}-"
        f"nf={current_params['n_filter_workers']}"
    )
    boom_consumer_log_fpath = f"logs/boom-{boom_config}/consumer.log"
    boom_scheduler_log_fpath = f"logs/boom-{boom_config}/scheduler.log"
    # To calculate BOOM wall time, take first timestamp from the consumer log
    # as the start and the last timestamp of the scheduler as the end
    if not os.path.isfile(boom_consumer_log_fpath):
        print(f"WARNING: {boom_consumer_log_fpath} does not exist")
        continue
    with open(boom_consumer_log_fpath) as f:
        line = f.readline()
        t1_b = pd.to_datetime(
            line.split()[2].replace("\x1b[2m", "").replace("\x1b[0m", "")
        )
    with open(boom_scheduler_log_fpath) as f:
        lines = f.readlines()
        line = lines[-3]
        t2_b = pd.to_datetime(
            line.split()[2].replace("\x1b[2m", "").replace("\x1b[0m", "")
        )
    current_params["start_time"] = t1_b
    current_params["end_time"] = t2_b
    boom_wall_time = t2_b - t1_b
    results_boom.append(current_params)

df_boom = pl.DataFrame(results_boom).with_columns(
    (pl.col("end_time") - pl.col("start_time"))
    .dt.total_seconds()
    .alias("wall_time_s")
)

# Kowalski only has one parameter name, so we don't need the product
param_name = list(kowalski_params.keys())[0]
param_vals = kowalski_params[param_name]
results_kowalski = []

for val in param_vals:
    res = {param_name: val}
    kowalski_config = f"n={val}"
    log_fpath = f"logs/kowalski-{kowalski_config}/supervisord.log"
    if not os.path.isfile(log_fpath):
        print(f"WARNING: {log_fpath} does not exist")
        continue
    # To calculate Kowalski wall time, just use the supervisord logs
    with open(log_fpath) as f:
        lines = f.readlines()
        for line in lines:
            if "alert-broker-ztf entered RUNNING state" in line:
                t1_k = pd.to_datetime("T".join(line.split()[:2]))
                break
        for line in lines:
            if "received SIGTERM indicating exit request" in line:
                t2_k = pd.to_datetime("T".join(line.split()[:2]))
                break
    # Print the total time of each, in minutes
    kowalski_wall_time = t2_k - t1_k  # type: ignore
    res["start_time"] = t1_k
    res["end_time"] = t2_k
    results_kowalski.append(res)

df_kowalski = pl.DataFrame(results_kowalski).with_columns(
    (pl.col("end_time") - pl.col("start_time"))
    .dt.total_seconds()
    .alias("wall_time_s")
)

# Write results to file for later analysis
os.makedirs("results", exist_ok=True)
df_boom.write_csv("results/boom.csv")
df_kowalski.write_csv("results/kowalski.csv")

# Fit a smooth response-surface model and predict the best worker split for each
# total
import numpy as np
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import (
    ConstantKernel,
    Matern,
    WhiteKernel,
)
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler

# Parameterize the problem by total workers and worker fractions
# This is smoother than fitting directly on the three raw counts under a
# fixed-sum constraint.
df_boom_model = df_boom.with_columns(
    n_workers=(
        pl.col("n_alert_workers")
        + pl.col("n_enrichment_workers")
        + pl.col("n_filter_workers")
    ),
    alert_frac=(
        pl.col("n_alert_workers")
        / (
            pl.col("n_alert_workers")
            + pl.col("n_enrichment_workers")
            + pl.col("n_filter_workers")
        )
    ),
    enrichment_frac=(
        pl.col("n_enrichment_workers")
        / (
            pl.col("n_alert_workers")
            + pl.col("n_enrichment_workers")
            + pl.col("n_filter_workers")
        )
    ),
)

model_features = ["n_workers", "alert_frac", "enrichment_frac"]
X = df_boom_model.select(model_features).to_numpy()
y = np.log(df_boom_model["wall_time_s"].to_numpy())

# Keep the GP smooth, but include an uncertainty penalty when selecting optima
# This helps avoid over-favoring edge allocations with unusually high filter
# fractions
kernel = ConstantKernel(1.0, (1e-3, 1e3)) * Matern(
    length_scale=[1.0, 1.0, 1.0], length_scale_bounds=(1e-2, 1e2), nu=2.5
) + WhiteKernel(noise_level=1e-3, noise_level_bounds=(1e-6, 1e1))

smooth_model = make_pipeline(
    StandardScaler(),
    GaussianProcessRegressor(
        kernel=kernel,
        normalize_y=True,
        n_restarts_optimizer=5,
        random_state=0,
    ),
)
smooth_model.fit(X, y)

# Penalize uncertain predictions so the recommended split stays in
# better-supported regions
uncertainty_penalty = 3


def generate_worker_splits(total_workers: int) -> list[tuple[int, int, int]]:
    if total_workers < 3:
        return []

    splits = []
    for n_alert_workers in range(1, total_workers - 1):
        for n_enrichment_workers in range(1, total_workers - n_alert_workers):
            n_filter_workers = (
                total_workers - n_alert_workers - n_enrichment_workers
            )
            if n_filter_workers >= 1:
                splits.append(
                    (
                        n_alert_workers,
                        n_enrichment_workers,
                        n_filter_workers,
                    )
                )
    return splits


def featurize_splits(
    splits: list[tuple[int, int, int]], total_workers: int
) -> np.ndarray:
    return np.array(
        [
            [
                total_workers,
                n_alert_workers / total_workers,
                n_enrichment_workers / total_workers,
            ]
            for n_alert_workers, n_enrichment_workers, _ in splits
        ]
    )


predicted_optima = []
gp = smooth_model.named_steps["gaussianprocessregressor"]
scaler = smooth_model.named_steps["standardscaler"]

for total_workers in range(3, 26):
    splits = generate_worker_splits(total_workers)
    split_features = featurize_splits(splits, total_workers)
    split_features_scaled = scaler.transform(split_features)
    predicted_log_wall, predicted_log_wall_std = gp.predict(
        split_features_scaled, return_std=True
    )
    conservative_log_wall = (
        predicted_log_wall + uncertainty_penalty * predicted_log_wall_std
    )
    predicted_wall_time = np.exp(predicted_log_wall)
    predicted_throughput = n_alerts / predicted_wall_time
    conservative_wall_time = np.exp(conservative_log_wall)
    best_index = int(np.argmin(conservative_log_wall))
    best_split = splits[best_index]
    predicted_optima.append(
        {
            "n_workers": total_workers,
            "n_alert": best_split[0],
            "n_enrich": best_split[1],
            "n_filter": best_split[2],
            "pred_wall_time_s": predicted_wall_time[best_index],
            "pred_alerts_per_second": predicted_throughput[best_index],
            "pred_log_wall_std": predicted_log_wall_std[best_index],
            "cons_wall_time_s": conservative_wall_time[best_index],
        }
    )

df_predicted_optima = pl.DataFrame(predicted_optima)

def get_data(
    system: Literal["boom", "kowalski"] = "boom",
    subsystem: Literal["scheduler", "consumer"] | None = None,
):
    fpath = f"logs/{system}-"
    if system == "boom":
        fpath += boom_cfg + f"/{subsystem}.stats.log"
    else:
        fpath += kowalski_config + "/ingester.stats.log"
    with open(fpath) as f:
        txt = f.read()
    txt = (
        txt.replace("\x1b[H", "")
        .replace("\x1b[K", "")
        .replace("\x1b[J", "")
        .split("\n")
    )
    data = []
    for row in txt:
        try:
            d = json.loads(row)
        except Exception:
            pass
        mem = d["MemUsage"].split(" / ")[0]
        if "KiB" in mem:
            mem = float(mem.split("KiB")[0]) / 1e3
        elif "MiB" in mem:
            mem = float(mem.split("MiB")[0])
        elif "GiB" in mem:
            mem = float(mem.split("GiB")[0]) * 1e3
        elif mem == "0B":
            mem = 0.0
        d["mem_mb"] = mem
        data.append(d)
    df = pl.DataFrame(data)
    return df
